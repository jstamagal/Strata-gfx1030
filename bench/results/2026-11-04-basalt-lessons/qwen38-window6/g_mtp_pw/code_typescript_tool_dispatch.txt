export interface JsonRpcRequest {
  jsonrpc: '2.0';
  method: string;
  params?: unknown;
  id?: string | number | null;
}

export interface JsonRpcResponse {
  jsonrpc: '2.0';
  result?: unknown;
  error?: {
    code: number;
    message: string;
    data?: unknown;
  };
  id: string | number | null;
}

type MethodHandler = (params: unknown, signal: AbortSignal) => Promise<unknown>;

interface ValidRequestInfo {
  isValid: boolean;
  isNotification: boolean;
  id: string | number | null;
  method?: string;
  params?: unknown;
  error?: { code: number; message: string };
}

function validateRequest(item: unknown): ValidRequestInfo {
  if (typeof item !== 'object' || item === null || Array.isArray(item)) {
    return { isValid: false, isNotification: true, id: null, error: { code: -32600, message: 'Invalid Request' } };
  }

  const obj = item as Record<string, unknown>;

  if (obj.jsonrpc !== '2.0') {
    return { isValid: false, isNotification: true, id: null, error: { code: -32600, message: 'Invalid Request' } };
  }

  const hasId = 'id' in obj;
  let id: string | number | null = null;
  if (hasId) {
    const rawId = obj.id;
    if (rawId === null || typeof rawId === 'string' || (typeof rawId === 'number' && Number.isFinite(rawId))) {
      id = rawId as string | number | null;
    } else {
      return { isValid: false, isNotification: true, id: null, error: { code: -32600, message: 'Invalid Request' } };
    }
  }

  const method = obj.method;
  if (typeof method !== 'string') {
    return { isValid: false, isNotification: !hasId, id, error: { code: -32600, message: 'Invalid Request' } };
  }

  const params = obj.params;

  return {
    isValid: true,
    isNotification: !hasId,
    id,
    method,
    params,
  };
}
