export interface JsonRpcRequest {
  jsonrpc: '2.0';
  method: string;
  params?: unknown;
  id?: string | number | null;
}

export interface JsonRpcResponse {
  jsonrpc: '2.0';
  result?: unknown;
  error?: {
    code: number;
    message: string;
    data?: unknown;
  };
  id: string | number | null;
}

type MethodHandler = (params: unknown, signal: AbortSignal) => Promise<unknown>;

interface ValidRequestInfo {
  isValid: boolean;
  isNotification: boolean;
  id: string | number | null | undefined;
  method?: string;
  params?: unknown;
}

function validateRequest(item: unknown): ValidRequestInfo {
  if (typeof item !== 'object' || item === null || Array.isArray(item)) {
    return { isValid: false, isNotification: false, id: undefined };
  }

  const obj = item as Record<string, unknown>;

  // Check jsonrpc field
  if (obj.jsonrpc !== '2.0') {
    return { isValid: false, isNotification: false, id: undefined };
  }

  // Check method field
  if (typeof obj.method !== 'string') {
    return { isValid: false, isNotification: false, id: undefined };
  }

  const method = obj.method;
  const params = obj.params;

  // Check id field
  let id: string | number | null | undefined;
  if ('id' in obj) {
    const rawId = obj.id;
    if (rawId === null || typeof rawId === 'string') {
      id = rawId;
    } else if (typeof rawId === 'number') {
      if (!Number.isFinite(rawId)) {
        return { isValid: false, isNotification: false, id: undefined };
      }
      id = rawId;
    } else {
      // Invalid id type
      return { isValid: false, isNotification: false, id: undefined };
    }
  } else {
    // Missing id means notification
    id = undefined;
  }

  const isNotification = id === undefined;

  return {
    isValid: true,
    isNotification,
    id,
    method,
    params,
  };
}

  };

function createSuccessResponse(result: unknown, id: string | number | null): JsonRpcResponse {
  return {
    jsonrpc: '2.0',
    result,
    id,
  };
}
