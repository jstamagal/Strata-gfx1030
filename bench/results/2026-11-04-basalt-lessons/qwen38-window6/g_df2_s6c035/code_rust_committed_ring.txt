use std::fmt;

#[derive(Debug, Clone, Copy, PartialEq)]
pub enum RingError {
    ZeroCapacity,
    InvalidCommitCount,
    StaleEpoch,
    RewindExposesOverwrittenRows,
}

impl fmt::Display for RingError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            RingError::ZeroCapacity => write!(f, "capacity must be non-zero"),
            RingError::InvalidCommitCount => write!(f, "invalid commit count"),
            RingError::StaleEpoch => write!(f, "stale epoch"),
            RingError::RewindExposesOverwrittenRows => write!(f, "rewind exposes overwritten rows"),
        }
    }
}

impl std::error::Error for RingError {}

#[derive(Debug, Clone, Copy)]
pub struct TokenRow {
    pub position: u64,
    pub token: u32,
    pub key: [f32; 4],
    pub value: [f32; 4],
}

#[derive(Debug, Clone, Copy)]
pub struct Snapshot {
    pub epoch: u64,
    pub next: u64,
}

#[derive(Debug)]
pub struct Ring {
    capacity: usize,
    rows: Vec<Option<TokenRow>>,
    epoch: u64,
    next: u64,
    staged_start: Option<u64>,
    staged_count: usize,
}

impl Ring {
    pub fn new(capacity: usize) -> Result<Self, RingError> {
        if capacity == 0 {
            return Err(RingError::ZeroCapacity);
        }
        Ok(Self {
            capacity,
            rows: vec![None; capacity],
            epoch: 0,
            next: 0,
            staged_start: None,
            staged_count: 0,
        })
    }

            return Ok(());

        let count = rows.len();

                return Err(RingError::InvalidCommitCount);
