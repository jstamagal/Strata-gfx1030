export interface JsonRpcRequest {
  jsonrpc: '2.0';
  method: string;
  params?: unknown;
  id?: string | number | null;
}

export interface JsonRpcResponse {
  jsonrpc: '2.0';
  result?: unknown;
  error?: {
    code: number;
    message: string;
    data?: unknown;
  };
  id: string | number | null;
}

type MethodHandler = (params: unknown, signal: AbortSignal) => Promise<unknown>;

interface ValidRequestInfo {
  index: number;
  isNotification: boolean;
  method: string;
  params: unknown;
  id: string | number | null;
}

interface InvalidRequestInfo {
  index: number;
  error: { code: number; message: string };
  id: string | number | null;
}

function isValidId(id: unknown): id is string | number | null {
  if (id === null) return true;
  if (typeof id === 'string') return true;
  if (typeof id === 'number') return Number.isFinite(id);
  return false;
}

function parseRequest(item: unknown, index: number): ValidRequestInfo | InvalidRequestInfo {
  // Check if it's an object
  if (item === null || typeof item !== 'object' || Array.isArray(item)) {
    return {
      index,
      error: { code: -32600, message: 'Invalid Request' },
      id: null,
    };
  }

  const obj = item as Record<string, unknown>;

  // Check jsonrpc field
  if (obj.jsonrpc !== '2.0') {
    return {
      index,
      error: { code: -32600, message: 'Invalid Request' },
      id: null,
    };
  }

  // Check method field
  if (typeof obj.method !== 'string') {
    return {
      index,
      error: { code: -32600, message: 'Invalid Request' },
      id: null,
    };
  }

  // Check id field if present
  let id: string | number | null = null;
  const hasId = 'id' in obj;
  if (hasId) {
    if (!isValidId(obj.id)) {
      return {
        index,
        error: { code: -32600, message: 'Invalid Request' },
        id: null,
      };
    }
    id = obj.id as string | number | null;
  }

  const isNotification = !hasId;

  return {
    index,
    isNotification,
    method: obj.method,
    params: obj.params,
    id,
  };
}
