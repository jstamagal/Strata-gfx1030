cmake_minimum_required(VERSION 3.21)
project(mtmd_embedding_reference C CXX)
set(LLAMA_DIR "" CACHE PATH "Independent ggml-org llama.cpp checkout")
if(NOT EXISTS "${LLAMA_DIR}/tools/mtmd/mtmd.h")
  message(FATAL_ERROR "Set LLAMA_DIR to the independent llama.cpp checkout")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_HIP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE ON CACHE BOOL "" FORCE)
set(GGML_OPENMP ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_DIR}" llama EXCLUDE_FROM_ALL)
add_executable(mtmd-embedding-reference mtmd_embedding_reference.cpp)
target_link_libraries(mtmd-embedding-reference PRIVATE mtmd llama)
target_compile_features(mtmd-embedding-reference PRIVATE cxx_std_17)
set_target_properties(mtmd-embedding-reference PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
